increment_percent_mapping={"Meets_Expectations":0.05,"Above_Expectations":0.10,"Below_Expectation":0.02,"Exceptional":0.15}

new_data=[]
for employee in employee_df.collect():
    increment = increment_percent_mapping[employee["Performance"]]
    new_salary = employee["Salary"] + (employee["Salary"] * increment)
    employee_Dict = employee.asDict()
    employee_Dict["new_salary"] = new_salary
    new_data.append(employee_Dict)

new_employee_df = spark.createDataFrame(new_data)
new_employee_df.show(5)

8.	Split the output files per (department, year)

            

        

        

            
In [56]:

            

                
unique_dy_combinations = new_employee_df.select("Department_ID","Year").distinct().collect() # df - list of row of object holding department and year
department_year_wise_records =[] 
# print(unique_dy_combinations)
for unique_comb in unique_dy_combinations:
    department = unique_comb["Department_ID"]
    year = unique_comb["Year"]
    employees_per_dy = new_employee_df.filter((new_employee_df["Department_ID"]==department) & (new_employee_df["Year"] == year)) # filter dataframe based on the unique department and year
    department_year_wise_records.append(employees_per_dy) # store each dataframe to a list 

# now i have the list of dataframe seperated by unique (department ,year)
for i,df in enumerate(department_year_wise_records,start=1):
    df.coalesce(1).write.mode("overwrite").option("header",True).csv(f"/home/sujay/Desktop/Python Learning/output/df_{i}")
